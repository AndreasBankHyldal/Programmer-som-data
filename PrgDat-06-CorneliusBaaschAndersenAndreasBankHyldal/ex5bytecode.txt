    LDARGS 1                            // Load command-line argument n
    CALL 1 L1                           // main(n)
    STOP                                // End execution

L1: INCSP 1                             // int r; [outer, offset 1]

    GETBP; CSTI 1; ADD                   // Address of outer r
    GETBP; CSTI 0; ADD; LDI              // Read n
    STI; INCSP -1                       // r = n;

    INCSP 1                             // int r; [inner, offset 2]

    GETBP; CSTI 0; ADD; LDI              // First argument: n
    GETBP; CSTI 2; ADD                   // Second argument: &inner_r
    CALL 2 L2                           // square(n, &r);
    INCSP -1                            // Discard dummy return value

    GETBP; CSTI 2; ADD; LDI              // Read inner r
    PRINTI; INCSP -1                    // print r; [inside nested block]

    INCSP -1                            // End nested block: discard inner r

    GETBP; CSTI 1; ADD; LDI              // Read outer r
    PRINTI; INCSP -1                    // print r; [outside nested block]

    INCSP -1                            // End main's block: discard outer r
    RET 0                               // Return from main

L2: GETBP; CSTI 1; ADD; LDI              // Read rp: destination address
    GETBP; CSTI 0; ADD; LDI              // Read i
    GETBP; CSTI 0; ADD; LDI              // Read i again
    MUL                                 // i * i
    STI; INCSP -1                       // *rp = i * i;

    INCSP 0                             // End square's block: no locals
    RET 1                               // Return from square
